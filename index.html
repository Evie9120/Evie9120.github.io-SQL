🔍 DIVINATION (SELECT)

SELECT * FROM table_name;
SELECT * FROM monsters;
👁️ Shows every column from the chosen table.
SELECT column1, column2 FROM table WHERE condition;
SELECT name, hp FROM monsters WHERE hp > 30;
⚡ Filters rows — only monsters with more than 30 HP appear.
SELECT * FROM chests WHERE gold_amount > 100;

✨ CONJURATION (INSERT)

INSERT INTO monsters VALUES ('name', hp, gold);
INSERT INTO monsters VALUES ('Troll', 65, 45);
🧌 Summons a new monster (name, HP, gold).
INSERT INTO chests VALUES ('location', gold_amount);

📦 Creates a treasure chest.
INSERT INTO heroes VALUES ('name', 'weapon');

⚡ TRANSMUTATION (UPDATE)

UPDATE table SET column = new_value WHERE condition;
UPDATE monsters SET hp = 100 WHERE name = 'Dragon';
❤️ Heals the Dragon to 100 HP.
UPDATE chests SET gold_amount = gold_amount * 2 WHERE location = 'Treasury';
UPDATE chests SET gold_amount = gold_amount * 2 WHERE location = 'Treasury';
💎 Doubles the gold in the Treasury chest.

🗑️ NECROMANCY (DELETE)

DELETE FROM table WHERE condition;
DELETE FROM monsters WHERE name = 'Skeleton';
💀 Banish the Skeleton from the dungeon.
DELETE FROM heroes WHERE weapon = 'Staff';
DELETE FROM heroes WHERE weapon = 'Staff';
🔮 Removes all heroes using a staff.
